\makeatletter
\newcommand{\mysortkey}[1]{%
  \compoundsortkeyaux#1-\@nil
}
\def\compoundsortkeyaux#1-#2\@nil{%
  \greeksort{#1}%
  \ifx\relax#2\relax
  \else
    -\compoundsortkeyaux#2\@nil
  \fi
}

% Return the first component of a compound key.
% alpha-x -> \alpha
% x-alpha -> x
\newcommand{\firstsymbol}[1]{%
  \firstsymbolaux\mysortkey{#1}-\@nil
}

\def\firstsymbolaux#1-#2\@nil{%
  \greeksymbol{#1}%
}

\makeatother

% glossary definition macro
\newcommand{\newsym}[3]{%
  \glsxtrnewsymbol[
    description={#3},
    sort={\mysortkey{#1}},
    group={\firstsymbol{#1}}
  ]{sym-#1}{\ensuremath{#2}}%
}
