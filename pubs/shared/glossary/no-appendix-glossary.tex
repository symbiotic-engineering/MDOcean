\def\debugglshypernumber#1{%
  \typeout{LOCATION ARG: [#1]}%
  \typeout{FIRST TOKEN: [\@car#1\@nil]}%
}

\newif\ifmainmatterglossary

\def\checkmainmatterlocations#1{%
  \edef\mainmatterfirstlocation{\@car#1\@nil}%
  \ifx\mainmatterfirstlocation\zerotoken
    \global\mainmatterglossarytrue
  \else
    \global\mainmatterglossaryfalse
  \fi
}

\def\zerotoken{0}

\newglossarystyle{mystyle}{%
  \setglossarystyle{list}%
  \setlist[description]{itemsep=0.15ex,parsep=0.15ex}

  \renewcommand*{\glossentry}[2]{%
    \mainmatterglossaryfalse
    \begingroup
      \def\debugglshypernumber####1{%
        \checkmainmatterlocations{####1}%
      }%
      \let\@xp@glshypernumber\debugglshypernumber
      ##2%
    \endgroup
    \ifmainmatterglossary
      \item
        \glsentryitem{##1}%
        \glstarget{##1}{\glossentryname{##1}}%
        \space
        \glossentrydesc{##1}%
    \fi
  }%
}

\makeatother
